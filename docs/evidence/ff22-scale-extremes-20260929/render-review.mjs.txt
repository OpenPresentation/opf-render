import {writeFile,mkdir} from 'node:fs/promises';
import {renderSvg,svgToPng} from './opf-render/dist/index.js';
const out=new URL('./visuals/',import.meta.url);await mkdir(out,{recursive:true});
const cases=[['subnormal','column',[Number.MIN_VALUE,2*Number.MIN_VALUE,4*Number.MIN_VALUE]],['maximum','column',[Number.MAX_VALUE/4,Number.MAX_VALUE/2,Number.MAX_VALUE]],['mixed-extremes','column',[-Number.MAX_VALUE,0,Number.MAX_VALUE]],['scatter-extremes','scatter',[-Number.MAX_VALUE,0,Number.MAX_VALUE]]];
for(const [name,type,values] of cases){
 const data={columns:type==='scatter'?['Category','X','Y']:['Category','Value'],rows:values.map((v,i)=>type==='scatter'?[['Negative','Zero','Positive'][i],v,v]:[['First','Second','Third'][i],v])};
 const document={slides:[{title:name,chart:{type,data}}]};
 const svg=renderSvg(document,{trace:true});await writeFile(new URL(name+'.svg',out),svg);await writeFile(new URL(name+'.opf.json',out),JSON.stringify(document,null,2)+'\n');
 await writeFile(new URL(name+'.png',out),await svgToPng(svg,{useBundledFonts:true,loadSystemFonts:false,scale:0.65}));
}
console.log('Four synthetic chart SVG/PNG review pairs; bundled fonts, system fonts disabled, estimated layout; no native or loaded-font browser claim.');
